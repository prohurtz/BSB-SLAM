cmake_minimum_required(VERSION 3.21)
project(BeyondSLAMTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
foreach(name angular_velocity floor_estimator floor_level stereo_frame calibration_tracker)
    add_executable(${name}_tests test_${name}.cpp)
    target_include_directories(${name}_tests PRIVATE ../Driver/openvr-master/headers)
    # Several tests use assert; retain assertions in Release builds.
    if(MSVC)
        target_compile_options(${name}_tests PRIVATE /UNDEBUG)
    else()
        target_compile_options(${name}_tests PRIVATE -UNDEBUG)
    endif()
    add_test(NAME ${name} COMMAND ${name}_tests)
endforeach()
if(WIN32)
    add_executable(camera_stream_tests test_camera_stream.cpp ../Driver/CameraStream.cpp)
    target_link_libraries(camera_stream_tests PRIVATE ws2_32 windowscodecs ole32)
    target_compile_options(camera_stream_tests PRIVATE /UNDEBUG)
    add_test(NAME camera_stream COMMAND camera_stream_tests)
endif()
